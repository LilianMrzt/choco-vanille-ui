import { Button, ToastProvider, useNotification, useToast } from '@choco-vanille/ui'

function Example() {
  const { showSuccess } = useToast()
  const { show: showNotification } = useNotification()

  return (
    <ToastProvider>
      <div style={{ display: 'flex', gap: '0.75rem', flexWrap: 'wrap' }}>
        <Button label="Show toast" onClick={() => showSuccess({ title: 'Changes saved', description: 'Your profile has been updated.' })} />
        <Button label="Show notification" onClick={() => showNotification({ variant: 'info', title: 'New version available', description: 'Reload when you are ready.' })} />
      </div>
    </ToastProvider>
  )
}
