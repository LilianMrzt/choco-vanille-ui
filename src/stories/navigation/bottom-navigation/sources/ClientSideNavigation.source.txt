import CircleIcon from '@iconify-icons/lucide/circle'
import SearchIcon from '@iconify-icons/lucide/search'
import UserIcon from '@iconify-icons/lucide/user'
import { useState } from 'react'
import { BottomNavigation } from '@choco-vanille/ui'

const Example = () => {
  const [route, setRoute] = useState('home')
  return (
    <div style={{ width: '100%' }}>
      <BottomNavigation aria-label="Application navigation" value={route}>
        <BottomNavigation.List>
          <BottomNavigation.Item value="home" label="Home" icon={CircleIcon} href="/" onClick={() => setRoute('home')} />
          <BottomNavigation.Item value="search" label="Search" icon={SearchIcon} href="/search" onClick={() => setRoute('search')} />
          <BottomNavigation.Item value="profile" label="Profile" icon={UserIcon} href="/profile" onClick={() => setRoute('profile')} />
        </BottomNavigation.List>
      </BottomNavigation>
      <p style={{ margin: '0.75rem 0 0', textAlign: 'center' }}>Current route: /{route}</p>
    </div>
  )
}
