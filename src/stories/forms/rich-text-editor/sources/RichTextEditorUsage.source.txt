import { useState } from 'react'
import { RichTextEditor, RichTextEditorField, type RichTextValue } from '@choco-vanille/ui'

const initialValue: RichTextValue = { type: 'doc', content: [] }

export const FieldExample = () => {
    const [value, setValue] = useState<RichTextValue>(initialValue)

    return <RichTextEditorField label="Description" placeholder="Start writing..." value={value} onValueChange={setValue} />
}
