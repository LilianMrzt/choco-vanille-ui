import { useState } from 'react'
import { Command, Button } from '@choco-vanille/ui'
// Replace these with icon components from your application.
import { SearchIcon } from './icons'

function ControlledCommandExample() {
  const [open, setOpen] = useState({{open}});

  return (
    <>
      <Button
        label="Toggle command palette (⌘K)"
        variant="contained"
        color="primary"
        onClick={() => setOpen(true)}
      />
      <Command open={open} onOpenChange={setOpen}>
        <Command.Content>
          <Command.Input placeholder="Search…" />
          <Command.List aria-label="Commands">
            <Command.Group label="Actions">
              <Command.Item icon={SearchIcon} shortcut="⌘E" onSelect={() => setOpen(false)}>
                Edit profile
              </Command.Item>
              <Command.Item icon={SearchIcon} shortcut="⌘S" onSelect={() => setOpen(false)}>
                Settings
              </Command.Item>
            </Command.Group>
            <Command.Empty>No results found.</Command.Empty>
          </Command.List>
        </Command.Content>
      </Command>
    </>
  );
}
