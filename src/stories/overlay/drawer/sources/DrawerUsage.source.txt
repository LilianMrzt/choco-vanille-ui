import { useState } from 'react'
import { Button, Drawer } from '@choco-vanille/ui'
import UserIcon from './UserIcon'

const Example = () => {
    const [open, setOpen] = useState(false)

    return (
        <>
            <Button label="Open drawer" onClick={() => setOpen(true)} />
            <Drawer aria-label="Settings" open={open} onOpenChange={setOpen}>
                <Drawer.Header><Button label="Close drawer" onClick={() => setOpen(false)} /></Drawer.Header>
                <Drawer.Body as="nav" aria-label="Main navigation">
                    <Drawer.Section title="Navigation"><Drawer.NavItem label="Profile" href="/profile" startIcon={UserIcon} /></Drawer.Section>
                </Drawer.Body>
            </Drawer>
        </>
    )
}
