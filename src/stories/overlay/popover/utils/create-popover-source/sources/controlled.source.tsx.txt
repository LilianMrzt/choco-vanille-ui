import { useState } from 'react'
import { Button, Popover } from '@choco-vanille/ui'

function Example() {
  const [open, setOpen] = useState({{open}})

  return (
    <Popover open={open} onOpenChange={setOpen} modal={{{modal}}}>
      <Popover.Trigger><Button label={{{triggerLabel}}} /></Popover.Trigger>
      <Popover.Content width={{{contentWidth}}} padding={{{contentPadding}}} aria-label="Options">
        <Popover.Item label={{{itemLabel}}} disabled={{{itemDisabled}}} destructive={{{itemDestructive}}} closeOnClick={{{itemCloseOnClick}}} />
      </Popover.Content>
    </Popover>
  )
}
